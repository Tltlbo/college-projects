#include <linux/module.h>
#include <linux/fs.h>
#include <linux/init.h>
#include <linux/kernel.h>
#include <linux/io.h>
#include <linux/slab.h>
#include <linux/gpio.h>
#include <mach/platform.h>
#include <asm/uaccess.h>

#define pump1_MAJOR	221          		// Major number of the device driver
#define pump1_NAME	“LED_DRIVER" 	// Name of the device driver 바꿔야 할걸?
#define GPIO_SIZE	256

char pump1_usage = 0;
static void *pump1_map;
volatile unsigned *pump1;

static int pump1_init();
static void pump1_exit();
static int pump1_release(struct inode *, struct file *);
static int pump1_open(struct inode *, struct file *);
static int pump1_write(struct file *, const char *, size_t, loff_t *);

static struct file_operations pump1_fops =
{
    .owner      = THIS_MODULE,
    .open        = pump1_open,
    .release    = pump1_release,
    .write        = pump1_write,
};

static int pump1_init(void)
{
    int result;
    result = register_chrdev(pump1_MAJOR, pump1_NAME, &pump1_fops);
    if (result < 0)
    {
printk(KERN_WARNING"Can't get any major!\n");
return result;
    }
}

static void pump1_exit(void)
{
    unregister_chrdev(pump1_MAJOR, pump1_NAME);
    printk(“pump1 module removed.\n");
}

module_init(pump1_init);      //Register the module and its functions.
module_exit(pump1_exit);
MODULE_LICENSE ("GPL");

static int pump1_open(struct inode *inode, struct file *mfile)
{
    if (pump1_usage != 0)
        return -EBUSY;
    pump1_usage = 1;			// pump driver usage check

    pump1_map = ioremap(GPIO_BASE, GPIO_SIZE); //0x3F20 0000
						     // Physical addr mapping
    if (!pump1_map)			 // Mapping error handling
    {
        printk(“error: mapping gpio memory”);
        iounmap(pump1_map);
        return –EBUSY;
    }
    pump1 = (volatile unsigned int *)pump1_map; // Set 4-byte pointer 
					          // to access the register(32bit)
    *(pump1 + 1) &= ~(0x07 << (3 * 7)); 	// select GPIO 17 and clear function
    *(pump1 + 1) |= (0x01 << (3 * 7)); 	// set GPIO17 as output mode

    return 0;
}

static int pump1_write(struct file *minode, const char *gdata, size_t length, loff_t *off_what)
{
    char tmp_buf;
    int result;

    result = copy_from_user(&tmp_buf, gdata, length) //1 byte data from
							  // application  pgm
	  if (result < 0)	// Error handling
    {
        printk(“Error: copy from user”);
        return result;
    }
    printk(“data from user: %d\n”, tmp_buf);
    if (tmp_buf == 0)
        *(pump1 + 7) = (0x01 << 17); 		// set GPIO17 as output 1
    else
        *(pump1 + 10) = (0x01 << 17); 		// clear GPIO17 as output 0
    return length;
}

static int pump1_release(struct inode *minode, struct file *mfile)
{
    led_usage = 0;

    if (pump1)   iounmap(pump1);

    return 0;
}
